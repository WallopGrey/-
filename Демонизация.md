Пример: BackEnd-сайт -> Python code. Исполняемый файл App.py. Если запустить его в терминале, а затем закрыть терминал, то BackEnd упадет, а процесс завершится. Решение: демонизация BackEnd'а - сделать из BackEnd'а системную службу (демон)

sudo nano /etc/systemd/system/resume-backend.service

[Unit]
    Description=Resume Counter Python Backend
    After=network.target
    
    [Service]
    # Укажите имя пользователя, от которого будет работать бэкенд
    User=vasya
    WorkingDirectory=/home/vasya/resume-backend
    # Путь к интерпретатору python внутри созданного venv
    ExecStart=/home/vasya/resume-backend/venv/bin/python3 app.py
    Restart=always
    
    [Install]
    WantedBy=multi-user.target

sudo systemctl daemon-reload
    sudo systemctl start resume-backend
    sudo systemctl enable resume-backend

sudo systemctl status resume-backend
